// CONFIGURACIÓN DE SEGURIDAD - AQUA LUAN const MASTER_USER = "supervisor"; const MASTER_PASS = "aqua2026"; const API_TOKEN = "aqua2026"; function doPost(e) { try { const data = JSON.parse(e.postData.contents); const accion = data.accion; // --- LOGIN SEGURO --- if (accion === 'login') { if (data.user === MASTER_USER && data.pass === MASTER_PASS) { return responder({ result: 'ok', message: 'Acceso concedido' }); } else { return responder({ result: 'error', error: 'Credenciales inválidas' }); } } // --- ACCIONES DE REGISTRO --- if (accion === 'registrar_pedido') return registrarPedido(data); if (accion === 'registrar_pago') return registrarPago(data); if (accion === 'registrar_gasto') return registrarGasto(data); if (accion === 'subir_pdf') return subirPDF(data); return responder({ result: 'error', error: 'Acción no reconocida' }); } catch (err) { return responder({ result: 'error', error: err.message }); } } function doGet(e) { // BLOQUEO DE SEGURIDAD: Solo responde si el token es correcto if (e.parameter.token !== API_TOKEN) { return responder({ result: 'error', error: 'No autorizado. Se requiere Token.' }); } const ss = SpreadsheetApp.getActiveSpreadsheet(); const hoja = ss.getSheetByName('GENERAL'); const datos = hoja.getDataRange().getValues(); if (datos.length <= 1) return responder({ result: 'ok', data: [] }); const encabezados = datos[0]; const filas = datos.slice(1).filter(f => f[0] !== ''); const registros = filas.map(f => { const obj = {}; encabezados.forEach((h, i) => obj[h] = f[i]); return obj; }); return responder({ result: 'ok', data: registros }); } // --- FUNCIONES DE APOYO (SIN CAMBIOS EN LÓGICA) --- function registrarPedido(d) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const hoja = ss.getSheetByName('GENERAL'); const hora = new Date().toLocaleTimeString('es-EC'); const productosRaw = d.productos ? d.productos.split(' | ') : []; productosRaw.forEach((prod, idx) => { const match = prod.match(/^(.+?)\s*\((\d+(?:\.\d+)?)\s*x\s*\$(\d+(?:\.\d+)?)\s*=\s*\$(\d+(?:\.\d+)?)\)$/); let nombre = prod, cantidad = '', precio = '', subtotal = ''; if (match) { nombre = match[1].trim(); cantidad = match[2]; precio = match[3]; subtotal = match[4]; } hoja.appendRow([d.fecha, d.dia, d.mes, d.empleado, d.formapago, d.nombre, d.telefono, d.direccion, nombre, cantidad, precio, subtotal, idx === 0 ? d.total : '', idx === 0 ? d.notas : '', idx === 0 ? d.latitud : '', idx === 0 ? d.longitud : '', idx === 0 ? d.gps_link : '', idx === 0 ? d.gps_prec : '', idx === 0 ? hora]); }); return responder({ result: 'ok' }); } function registrarPago(d) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const hoja = ss.getSheetByName('GENERAL'); hoja.appendRow([d.fecha, '', '', d.empleado, d.forma, d.cliente, '', '', '', '', '', '', d.monto, d.notas, '', '', '', '', new Date().toLocaleTimeString('es-EC')]); return responder({ result: 'ok' }); } function registrarGasto(d) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const hoja = ss.getSheetByName('GENERAL'); hoja.appendRow([d.fecha, '', '', d.empleado, d.categoria, d.desc, '', '', '', '', '', '', '-' + d.monto, d.ref, '', '', '', '', new Date().toLocaleTimeString('es-EC')]); return responder({ result: 'ok' }); } function subirPDF(d) { const nombreCarpeta = 'AQUA LUAN - Pedidos PDF'; let carpeta; const carpetas = DriveApp.getFoldersByName(nombreCarpeta); carpeta = carpetas.hasNext() ? carpetas.next() : DriveApp.createFolder(nombreCarpeta); const blob = Utilities.newBlob(Utilities.base64Decode(d.pdfBase64), 'application/pdf', d.nombreArchivo); const archivo = carpeta.createFile(blob); archivo.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); return responder({ result: 'ok', fileUrl: archivo.getUrl() }); } function responder(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); }
